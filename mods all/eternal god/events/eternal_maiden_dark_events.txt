# The Servant Who Knew Too Much
eternal_maiden.0407 = {
	type = character_event
	title = eternal_maiden.0407.t
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { scope:maiden_speaker = { is_female = yes } }
				desc = eternal_maiden.0407.desc_f
			}
			desc = eternal_maiden.0407.desc_m
		}
	}
	theme = intrigue
	override_background = { reference = eternal_bloodline_assassin }
	left_portrait = { character = root animation = personality_vengeful }
	right_portrait = { character = scope:maiden_speaker animation = fear }

	trigger = {
		has_trait = holy_maiden
		any_courtier = { is_adult = yes is_imprisoned = no }
		NOT = { has_character_flag = eternal_maiden_0407_cd }
	}

	immediate = {
		add_character_flag = { flag = eternal_maiden_0407_cd years = 15 }
		random_courtier = {
			limit = { is_adult = yes is_imprisoned = no }
			save_scope_as = maiden_speaker
		}
	}

	option = {
		name = eternal_maiden.0407.a
		add_dread = 25
		eternal_add_suspicion = { A = -20 B = -10 }
		scope:maiden_speaker = { add_opinion = { target = root modifier = eternal_maiden_shamed_opinion } }
	}

	option = {
		name = eternal_maiden.0407.b
		add_hook = { type = favor_hook target = scope:maiden_speaker }
	}

	option = {
		name = eternal_maiden.0407.c
		scope:maiden_speaker = { add_opinion = { target = root modifier = eternal_maiden_devoted_opinion } }
		eternal_add_suspicion = { A = 5 B = 2 }
	}
}
