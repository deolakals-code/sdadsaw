# Story chain for the Holy Maiden's war. 0001 starts on declaration (eternal_holy_maiden_war on_declaration);
# 0002-0004 follow while the war lasts; 0005-0007 close the story on victory / white peace / defeat.

eternal_holywar.0001 = {
	type = character_event
	title = eternal_holywar.0001.t
	desc = eternal_holywar.0001.desc
	theme = faith
	override_background = { reference = eternal_bloodline_worship }
	left_portrait = { character = root animation = personality_zealous }

	option = {
		name = eternal_holywar.0001.a
		add_piety = 150
		add_character_flag = { flag = eternal_holywar_mercy years = 2 }
		trigger_event = { id = eternal_holywar.0002 days = { 10 20 } }
	}

	option = {
		name = eternal_holywar.0001.b
		add_dread = 40
		add_prestige = 50
		add_character_flag = { flag = eternal_holywar_fire years = 2 }
		trigger_event = { id = eternal_holywar.0002 days = { 10 20 } }
	}

	option = {
		name = eternal_holywar.0001.c
		add_prestige = 150
		trigger_event = { id = eternal_holywar.0002 days = { 10 20 } }
	}
}

# Hidden upkeep: every ten days of a Holy Maiden war, clears mercenary payment claims of the hired adventurers.
eternal_holywar.0100 = {
	hidden = yes

	immediate = {
		if = {
			limit = {
				exists = var:eternal_war_ref
				is_at_war = yes
			}
			var:eternal_war_ref = { save_scope_as = war }
			eternal_clear_merc_pay_effect = yes
			trigger_event = { id = eternal_holywar.0100 days = 10 }
		}
		else_if = {
			limit = { exists = var:eternal_war_ref }
			remove_variable = eternal_war_ref
		}
	}
}
