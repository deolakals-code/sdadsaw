namespace = eternal_edict

# Edict commands: activities the Holy Maiden can order of a ruler who is Under the Edict.
# Fired from common/character_interactions/eternal_edict_interactions.txt. 0010 and 0011 are adult-only.

# Edict 0001
eternal_edict.0001 = {
	type = character_event
	title = eternal_edict.0001.t
	desc = eternal_edict.0001.desc
	theme = realm
	override_background = { reference = eternal_bloodline_worship }
	left_portrait = { character = root animation = personality_rational }
	right_portrait = { character = scope:summon_guest animation = idle }

	trigger = {
		exists = var:summon_target
		var:summon_target = { is_alive = yes }
	}

	immediate = {
		var:summon_target ?= { save_scope_as = summon_guest }
	}

	option = {
		name = eternal_edict.0001.a
		scope:summon_guest = { add_opinion = { target = root modifier = eternal_charmed_opinion } }
		add_prestige = 50
	}

	option = {
		name = eternal_edict.0001.b
		add_dread = 30
		scope:summon_guest = { add_stress = 15 }
	}

	option = {
		name = eternal_edict.0001.c
		random_list = {
			70 = {
				add_hook = { type = favor_hook target = scope:summon_guest }
			}
			30 = {
				scope:summon_guest = { add_opinion = { target = root modifier = eternal_maiden_shamed_opinion } }
			}
		}
	}
}
