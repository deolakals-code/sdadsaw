namespace = eternal_maiden

# Holy Maiden 3/4: the court and its rivals (0300-0309). Fired from holy_maiden_events_yearly.

# The Jealous Rival
eternal_maiden.0300 = {
	type = character_event
	title = eternal_maiden.0300.t
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { scope:maiden_speaker = { is_female = yes } }
				desc = eternal_maiden.0300.desc_f
			}
			desc = eternal_maiden.0300.desc_m
		}
	}
	theme = intrigue
	override_background = { reference = eternal_bloodline_worship }
	left_portrait = { character = root animation = personality_rational }
	right_portrait = { character = scope:maiden_speaker animation = idle }

	trigger = {
		has_trait = holy_maiden
		any_courtier = { is_adult = yes is_imprisoned = no }
		NOT = { has_character_flag = eternal_maiden_0300_cd }
	}

	immediate = {
		add_character_flag = { flag = eternal_maiden_0300_cd years = 15 }
		random_courtier = {
			limit = { is_adult = yes is_imprisoned = no }
			save_scope_as = maiden_speaker
		}
	}

	option = {
		name = eternal_maiden.0300.a
		add_dread = 20
		scope:maiden_speaker = { add_opinion = { target = root modifier = eternal_maiden_shamed_opinion } }
	}

	option = {
		name = eternal_maiden.0300.b
		add_piety = 100
		scope:maiden_speaker = { add_opinion = { target = root modifier = eternal_maiden_devoted_opinion } }
	}

	option = {
		name = eternal_maiden.0300.c
		add_prestige = 100
		eternal_add_suspicion = { A = 6 B = 3 }
		scope:maiden_speaker = { add_opinion = { target = root modifier = eternal_maiden_shamed_opinion } }
	}
}
