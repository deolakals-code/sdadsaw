# Royal Summons: the Holy Maiden calls any ruler to her court by carriage (interaction eternal_royal_summons_interaction).
# 0001 target decides, 0002 refusal, 0010 arrival, 0100 what the Maiden wills, 1100-5100 results, 0200 departure, 1200-5200 results.
# Scopes: the Maiden holds var:summon_target; the summoned ruler holds var:summoned_by.

# The summons arrives
eternal_summons.0001 = {
	type = character_event
	title = eternal_summons.0001.t
	desc = eternal_summons.0001.desc
	theme = intrigue
	override_background = { reference = eternal_bloodline_assassin }
	left_portrait = { character = root animation = fear }

	trigger = {
		exists = var:summoned_by
	}

	option = {
		name = eternal_summons.0001.a
		ai_chance = { base = 90 }
		hidden_effect = {
			start_travel_plan = {
				destination = var:summoned_by.capital_province
				on_arrival_on_action = eternal_summons_arrive
				on_travel_planner_cancel_on_action = eternal_summons_cancel
				on_arrival_destinations = last
				return_trip = no
				players_use_planner = no
			}
		}
	}

	option = {
		name = eternal_summons.0001.b
		ai_chance = { base = 10 }
		var:summoned_by ?= { save_scope_as = summoner }
		scope:summoner = { trigger_event = eternal_summons.0002 }
		hidden_effect = { death = { death_reason = death_execution killer = var:summoned_by } }
	}
}

# Result 3100
eternal_summons.3100 = {
	type = character_event
	title = eternal_summons.3100.t
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { scope:summon_guest = { is_female = yes } }
				desc = eternal_summons.3100.desc_f
			}
			desc = eternal_summons.3100.desc_m
		}
	}
	theme = intrigue
	override_background = { reference = eternal_bloodline_awakening }
	left_portrait = { character = root animation = happiness }
	right_portrait = { character = scope:summon_guest animation = happiness }

	trigger = {
		exists = var:summon_target
		var:summon_target = { is_alive = yes }
	}

	immediate = {
		var:summon_target ?= { save_scope_as = summon_guest }
	}

	option = {
		name = eternal_summons_result_ok
		set_relation_lover = scope:summon_guest
		add_hook = { type = favor_hook target = scope:summon_guest }
		add_stress = -20
		if = {
			limit = {
				is_female = yes
				is_pregnant = no
				can_have_children = yes
				scope:summon_guest = { is_male = yes can_have_children = yes }
			}
			random = {
				chance = 40
				make_pregnant = { father = scope:summon_guest number_of_children = 1 }
			}
		}
	}
}
