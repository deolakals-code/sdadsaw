# Royal Summons: the Holy Maiden calls any ruler to her court by carriage (interaction eternal_royal_summons_interaction).
# 0001 target decides, 0002 refusal, 0010 arrival, 0100 what the Maiden wills, 1100-5100 results, 0200 departure, 1200-5200 results.
# Scopes: the Maiden holds var:summon_target; the summoned ruler holds var:summoned_by.

# The summons arrives
eternal_summons.0001 = {
	type = character_event
	title = eternal_summons.0001.t
	desc = eternal_summons.0001.desc
	theme = intrigue
	override_background = { reference = eternal_bloodline_assassin }
	left_portrait = { character = root animation = fear }

	trigger = {
		exists = var:summoned_by
	}

	option = {
		name = eternal_summons.0001.a
		ai_chance = { base = 90 }
		eternal_summons_depart_effect = yes
	}

	option = {
		name = eternal_summons.0001.b
		ai_chance = { base = 10 }
		var:summoned_by ?= { save_scope_as = summoner }
		scope:summoner = { trigger_event = eternal_summons.0002 }
		hidden_effect = { death = { death_reason = death_execution killer = var:summoned_by } }
	}
}

eternal_summons.0050 = {
	hidden = yes

	immediate = {
		if = {
			limit = {
				exists = var:summon_target
				NOT = { has_variable = eternal_summon_arrived }
			}
			var:summon_target = {
				if = {
					limit = { is_alive = yes }
					current_travel_plan ?= { abort_travel_plan = yes }
					set_location = root.location
				}
			}
			if = {
				limit = { var:summon_target = { is_alive = yes } }
				trigger_event = eternal_summons.0010
			}
			else = {
				remove_variable = summon_target
			}
		}
	}
}
