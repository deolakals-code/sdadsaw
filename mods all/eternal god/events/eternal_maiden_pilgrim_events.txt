# The Blind Pilgrim
eternal_maiden.0100 = {
	type = character_event
	title = eternal_maiden.0100.t
	desc = eternal_maiden.0100.desc
	theme = faith
	override_background = { reference = eternal_bloodline_worship }
	left_portrait = { character = root animation = personality_zealous }

	trigger = {
		has_trait = holy_maiden
		NOT = { has_character_flag = eternal_maiden_0100_cd }
	}

	immediate = {
		add_character_flag = { flag = eternal_maiden_0100_cd years = 15 }
	}

	option = {
		name = eternal_maiden.0100.a
		trigger_event = { id = eternal_maiden.1100 }
	}

	option = {
		name = eternal_maiden.0100.b
		trigger_event = { id = eternal_maiden.2100 }
	}

	option = {
		name = eternal_maiden.0100.c
		trigger_event = { id = eternal_maiden.3100 }
	}
}

# The Blind Pilgrim - epilogue
eternal_maiden.5100 = {
	type = character_event
	title = eternal_maiden.5100.t
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { has_character_flag = eternal_maiden_0100_echo_a }
				desc = eternal_maiden.5100.desc_a
			}
			triggered_desc = {
				trigger = { has_character_flag = eternal_maiden_0100_echo_b }
				desc = eternal_maiden.5100.desc_b
			}
			triggered_desc = {
				trigger = { has_character_flag = eternal_maiden_0100_echo_c }
				desc = eternal_maiden.5100.desc_c
			}
		}
	}
	theme = faith
	override_background = { reference = eternal_bloodline_worship }
	left_portrait = { character = root animation = personality_zealous }

	option = {
		name = eternal_maiden_echo_ok
		if = {
			limit = { has_character_flag = eternal_maiden_0100_echo_a }
			add_prestige = 150
			add_character_modifier = { modifier = eternal_maiden_miracle_modifier years = 2 }
			remove_character_flag = eternal_maiden_0100_echo_a
		}
		if = {
			limit = { has_character_flag = eternal_maiden_0100_echo_b }
			add_prestige = 250
			add_dread = 10
			remove_character_flag = eternal_maiden_0100_echo_b
		}
		if = {
			limit = { has_character_flag = eternal_maiden_0100_echo_c }
			add_piety = 200
			add_stress = -20
			remove_character_flag = eternal_maiden_0100_echo_c
		}
	}
}
