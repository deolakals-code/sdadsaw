# Holy Maiden 2/4: clergy, heresy and the head of faith (0200-0209). Each option opens its own result event; some have a delayed epilogue.

# A Letter from the Head of Faith
eternal_maiden.0200 = {
	type = character_event
	title = eternal_maiden.0200.t
	desc = eternal_maiden.0200.desc
	theme = faith
	override_background = { reference = eternal_bloodline_worship }
	left_portrait = { character = root animation = personality_rational }
	right_portrait = { character = scope:maiden_pontiff animation = idle }

	trigger = {
		has_trait = holy_maiden
		exists = faith.religious_head
		NOT = { faith.religious_head = root }
		NOT = { has_character_flag = eternal_maiden_0200_cd }
	}

	immediate = {
		add_character_flag = { flag = eternal_maiden_0200_cd years = 15 }
		faith.religious_head = { save_scope_as = maiden_pontiff }
	}

	option = {
		name = eternal_maiden.0200.a
		set_variable = { name = maiden_target value = scope:maiden_pontiff }
		trigger_event = { id = eternal_maiden.1200 }
	}

	option = {
		name = eternal_maiden.0200.b
		set_variable = { name = maiden_target value = scope:maiden_pontiff }
		trigger_event = { id = eternal_maiden.2200 }
	}

	option = {
		name = eternal_maiden.0200.c
		set_variable = { name = maiden_target value = scope:maiden_pontiff }
		trigger_event = { id = eternal_maiden.3200 }
	}
}

# A Letter from the Head of Faith - epilogue
eternal_maiden.5200 = {
	type = character_event
	title = eternal_maiden.5200.t
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { has_character_flag = eternal_maiden_0200_echo_a }
				desc = eternal_maiden.5200.desc_a
			}
			triggered_desc = {
				trigger = { has_character_flag = eternal_maiden_0200_echo_b }
				desc = eternal_maiden.5200.desc_b
			}
			triggered_desc = {
				trigger = { has_character_flag = eternal_maiden_0200_echo_c }
				desc = eternal_maiden.5200.desc_c
			}
		}
	}
	theme = faith
	override_background = { reference = eternal_bloodline_worship }
	left_portrait = { character = root animation = personality_rational }

	option = {
		name = eternal_maiden_echo_ok
		if = {
			limit = { has_character_flag = eternal_maiden_0200_echo_a }
			add_piety = 150
			remove_character_flag = eternal_maiden_0200_echo_a
		}
		if = {
			limit = { has_character_flag = eternal_maiden_0200_echo_b }
			add_prestige = 150
			add_dread = 10
			remove_character_flag = eternal_maiden_0200_echo_b
		}
		if = {
			limit = { has_character_flag = eternal_maiden_0200_echo_c }
			add_piety = 100
			add_stress = -10
			remove_character_flag = eternal_maiden_0200_echo_c
		}
	}
}
