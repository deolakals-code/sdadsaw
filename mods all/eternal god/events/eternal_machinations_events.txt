		trigger_event = { id = eternal_machinations.0301 days = 20 }
	}
}

eternal_machinations.0301 = {
	type = character_event
	title = eternal_machinations.0301.t
	desc = eternal_machinations.0301.desc
	theme = war
	override_background = { reference = eternal_bloodline_assassin }
	left_portrait = { character = root animation = personality_vengeful }

	immediate = {
		var:eternal_m_feud_target ?= {
			save_scope_as = feud_a
			random_neighboring_and_across_water_top_liege_realm_owner = {
				limit = {
					NOT = { this = root }
					NOT = { is_allied_to = scope:feud_a }
					NOT = { has_truce = scope:feud_a }
				}
				save_scope_as = feud_b
			}
		}
	}

	option = {
		name = eternal_machinations.0301.a
		scope:feud_a ?= {
			add_opinion = { target = scope:feud_b modifier = eternal_slandered_opinion }
		}
		scope:feud_b ?= {
			add_opinion = { target = scope:feud_a modifier = eternal_slandered_opinion }
			scope:feud_a.capital_county ?= { save_scope_as = feud_claim }
			if = {
				limit = { exists = scope:feud_claim }
				add_pressed_claim = scope:feud_claim
				start_war = {
					cb = claim_cb
					target = scope:feud_a
					target_title = scope:feud_claim
					claimant = scope:feud_b
				}
			}
		}
		add_character_modifier = { modifier = eternal_shadow_sovereign_modifier years = 10 }
		remove_variable = eternal_m_feud_target
	}
}
