# Test of Loyalty: a vassal must kneel
eternal_tyrant.0003 = {
	type = character_event
	title = eternal_tyrant.0003.t
	desc = eternal_tyrant.0003.desc
	theme = realm
	override_background = { reference = eternal_bloodline_worship }
	left_portrait = { character = root animation = personality_bold }
	right_portrait = { character = scope:tyrant_vassal animation = fear }

	trigger = {
		NOT = { has_character_flag = eternal_tyrant_0003_cd }
		any_vassal = { is_adult = yes is_imprisoned = no }
	}

	immediate = {
		add_character_flag = { flag = eternal_tyrant_0003_cd years = 10 }
		random_vassal = {
			limit = { is_adult = yes is_imprisoned = no }
			save_scope_as = tyrant_vassal
		}
	}

	option = {
		name = eternal_tyrant.0003.a
		add_hook = { type = favor_hook target = scope:tyrant_vassal }
		scope:tyrant_vassal = {
			add_opinion = { target = root modifier = eternal_tyrant_sworn_opinion }
		}
	}
	option = {
		name = eternal_tyrant.0003.b
		add_dread = 35
		scope:tyrant_vassal = {
			add_opinion = { target = root modifier = eternal_tyrant_terrified_opinion }
		}
		stress_impact = { compassionate = minor_stress_impact_gain }
	}
	option = {
		name = eternal_tyrant.0003.c
		add_prestige = 100
		scope:tyrant_vassal = {
			add_opinion = { target = root modifier = eternal_tyrant_sworn_opinion }
		}
	}
}
