eternal_edict_lesson_interaction = {
	category = interaction_category_eternal_edict
	icon = icon_scheme_befriend
	interface_priority = 79
	desc = eternal_edict_lesson_interaction_desc
	auto_accept = yes
	cooldown = { days = 15 }
	is_shown = {
		scope:actor = {
			has_trait = holy_maiden
			exists = var:summon_target
			var:summon_target = scope:recipient
		}
		scope:recipient = { has_character_modifier = eternal_under_edict_modifier }
	}
	is_valid_showing_failures_only = {
		scope:actor = { is_imprisoned = no }
	}
	on_accept = {
		scope:actor = { trigger_event = eternal_edict.0001 }
	}
	ai_accept = { base = 100 }
	ai_potential = { always = no }
	ai_will_do = { base = 0 }
}
