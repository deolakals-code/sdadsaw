eternal_decree_excommunicate_interaction = {
	category = interaction_category_eternal_realm
	icon = icon_scheme_befriend
	interface_priority = 75
	desc = eternal_decree_excommunicate_interaction_desc
	auto_accept = yes
	cooldown = { days = 90 }
	is_shown = {
		scope:actor = { has_trait = holy_maiden }
		scope:recipient != scope:actor
		scope:recipient = { is_alive = yes }
	}
	is_valid_showing_failures_only = {
		scope:recipient = { NOT = { has_trait = excommunicated } }

	on_accept = {
		scope:recipient = {
			add_trait_force_tooltip = excommunicated
			add_opinion = { target = scope:actor modifier = eternal_exposed_opinion }
		}
	}
	ai_accept = { base = 100 }
	ai_potential = { always = no }
	ai_will_do = { base = 0 }
}

			trigger_event = eternal_summons.0001
		}
	}
	ai_accept = { base = 100 }
	ai_potential = { always = no }
