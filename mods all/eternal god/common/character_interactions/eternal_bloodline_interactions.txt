eternal_annihilate_interaction = {
	category = interaction_category_eternal_realm
	icon = icon_scheme_murder
	interface_priority = 90
	desc = eternal_annihilate_interaction_desc
	auto_accept = yes
	cooldown = { days = 30 }

	is_shown = {
		scope:actor = { has_trait = eternal_bloodline }
		scope:recipient != scope:actor
		scope:recipient = { is_alive = yes }
	}

	is_valid_showing_failures_only = {
		scope:actor = {
			is_imprisoned = no
			NOT = { has_variable = eternal_annihilate_target }
		}
	}

	on_accept = {
		scope:actor = {
			set_variable = { name = eternal_annihilate_target value = scope:recipient }
			trigger_event = eternal_annihilate.0001
		}
	}

	ai_accept = { base = 100 }
	ai_potential = { always = no }
	ai_will_do = { base = 0 }
}
