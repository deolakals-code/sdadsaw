eternal_rumor_interaction = {
	category = interaction_category_eternal_realm
	icon = icon_scheme_murder
	interface_priority = 80
	desc = eternal_rumor_interaction_desc
	auto_accept = yes
	cooldown = { days = 60 }
	is_shown = {
		scope:actor = { has_trait = eternal_bloodline }
		scope:recipient = { is_ruler = yes is_independent_ruler = yes }
		scope:recipient != scope:actor
	}
	on_accept = {
		scope:actor = {
			set_variable = { name = eternal_m_rumor_target value = scope:recipient }
			trigger_event = eternal_machinations.0100
		}
	}
	ai_accept = { base = 100 }
	ai_potential = { always = no }
	ai_will_do = { base = 0 }
}
