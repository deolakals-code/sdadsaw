# Hooks into game on_actions only through on_actions lists, so the game's own
# effects for these on_actions are kept.
on_birth_child = {
	on_actions = { eternal_bloodline_on_birth_child }
}

on_ruler_designer_finished = {
	on_actions = { eternal_bloodline_on_ruler_designer_finished }
}

on_combat_end_winner = {
	on_actions = { eternal_bloodline_on_combat_end }
}

on_combat_end_loser = {
	on_actions = { eternal_bloodline_on_combat_end }
}

yearly_playable_pulse = {
	on_actions = { eternal_bloodline_yearly }
}

# AI characters must not keep the trait; paired with random_creation_weight = 0.
eternal_bloodline_on_birth_child = {
	trigger = { scope:child = { has_trait = eternal_bloodline } }
	effect = {
		scope:child = {
			if = {
				limit = {
					has_trait = eternal_bloodline
					is_ai = yes
					NOT = { has_character_flag = eternal_bloodline_chosen }
				}
				remove_trait = eternal_bloodline
			}
		}
	}
}

# Traits have no on_add, so the awakening runs from the ruler designer.
eternal_bloodline_on_ruler_designer_finished = {
	trigger = { has_trait = eternal_bloodline }
	effect = {
		if = {
			limit = { has_trait = eternal_bloodline }
			add_character_flag = eternal_bloodline_chosen
			add_character_flag = immune_to_disease
			eternal_bloodline_purge_injuries = yes
			eternal_bloodline_grant_regiment = yes
			trigger_event = { id = eternal_bloodline.0001 days = 1 }
		}
	}
}

# Root = combat side
eternal_bloodline_on_combat_end = {
	trigger = {
		OR = {
			side_primary_participant ?= { has_trait = eternal_bloodline }
			side_commander ?= { has_trait = eternal_bloodline }
		}
	}
	effect = {
		side_primary_participant ?= { eternal_bloodline_purge_injuries = yes }
		side_commander ?= { eternal_bloodline_purge_injuries = yes }
	}
}

# Fallback for characters who got the trait by console/inheritance.
eternal_bloodline_yearly = {
	trigger = { has_trait = eternal_bloodline }
	effect = {
		add_character_flag = immune_to_disease
		eternal_bloodline_purge_injuries = yes
		eternal_bloodline_grant_regiment = yes
	}
	random_events = {
		chance_of_no_event = 90
		1 = eternal_bloodline.0002
		4 = eternal_bloodline.0003
	}
}
