yearly_playable_pulse = {
	on_actions = { eternal_tyrant_yearly }
}

eternal_tyrant_yearly = {
	trigger = {
		has_trait = eternal_bloodline
		is_ruler = yes
		is_adult = yes
	}
	random_events = {
		chance_of_no_event = 85
		100 = eternal_tyrant.0001
		100 = eternal_tyrant.0002
		100 = eternal_tyrant.0003
		100 = eternal_tyrant.0004
		100 = eternal_tyrant.0005
		100 = eternal_tyrant.0006
	}
}
