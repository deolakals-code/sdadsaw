on_birth_child = {
	on_actions = { holy_maiden_on_birth_child }
}

on_ruler_designer_finished = {
	on_actions = { holy_maiden_on_ruler_designer_finished }
}

yearly_playable_pulse = {
	on_actions = { holy_maiden_yearly }
}

holy_maiden_on_birth_child = {
	effect = {
		scope:child = {
			if = {
				limit = { has_trait = holy_maiden NOT = { has_character_flag = holy_maiden_chosen } }
				remove_trait = holy_maiden
			}
		}
	}
}

holy_maiden_on_ruler_designer_finished = {
	effect = {
		if = {
			limit = { has_trait = holy_maiden }
			add_character_flag = holy_maiden_chosen
		}
	}
}

holy_maiden_yearly = {
	effect = {
		if = {
			limit = {
				has_trait = holy_maiden
				is_ai = yes
				NOT = { has_character_flag = holy_maiden_chosen }
			}
			remove_trait = holy_maiden
		}
	}
}
