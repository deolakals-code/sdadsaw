# Outbound leg finished: we are at the target's location.
# Own realm -> eternal_carriage.0001 (touring our lands), other realm -> eternal_carriage.0002 (state visit)
eternal_carriage_arrive = {
	effect = {
		if = {
			limit = {
				has_variable = eternal_carriage_target
				var:eternal_carriage_target ?= { is_alive = yes }
			}
			if = {
				limit = { var:eternal_carriage_target.top_liege = root.top_liege }
				trigger_event = eternal_carriage.0001
			}
			else = {
				trigger_event = eternal_carriage.0002
			}
		}
		else = {
			eternal_carriage_cleanup_effect = yes
		}
	}
}

# Return leg finished: we are home. Make sure the guest actually arrived.
eternal_carriage_home = {
	effect = {
		eternal_carriage_bring_guest_effect = yes
		eternal_carriage_cleanup_effect = yes
	}
}

eternal_carriage_cancel = {
	effect = {
		eternal_carriage_cleanup_effect = yes
	}
}
