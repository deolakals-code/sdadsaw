# Holy Maiden wars never charge the Maiden for hired adventurers (Roads to Power mercenary payouts).
# The payout letter needs var:joined_as_mercenary on the adventurer, so this clears it (and the employer's
# hired_mercenaries entry, as the base game does after paying). Needs scope:war.
eternal_clear_merc_pay_effect = {
	scope:war = {
		every_war_participant = {
			limit = { exists = var:joined_as_mercenary }
			save_scope_as = eternal_merc
			var:joined_as_mercenary = {
				remove_list_variable = { name = hired_mercenaries target = scope:eternal_merc }
			}
			remove_variable = joined_as_mercenary
		}
	}
}
